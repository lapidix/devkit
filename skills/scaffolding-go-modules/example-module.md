# Example module: `internal/modules/user`

One complete module that follows the Go server rules: huma on `net/http`, sqlc with pgx, goose migrations, testify. The module path is `example.com/server`. It builds, passes golangci-lint with the configuration in `rules/go/conventions.md`, the unit tests, the database tests against PostgreSQL, the architecture test and govulncheck, and its routes answer when the server runs.

The project already has, once: the three `internal/shared` packages at the top, `internal/common/apiresponse` and `internal/common/middleware` (`rules/go-server/http.md`), `internal/libs/requestinfo` and `logger`, `internal/infrastructure/database` with `dbtest` (`rules/database/sqlc.md`), `tools/go.mod` with sqlc, goose and govulncheck, and an `auth` module whose root package re-exports `RevokeAllTokens`. `sqlcgen/` is generated by `make generate` and not shown. Creating users (and so `ids.New()` in a use case, and a `201` route) is left out; the repository test shows `ids.New()`.

```go
	items := make([]R, 0, len(p.Items))
```

```go
package domain

type Nickname struct {
	value string
}

func NewNickname(raw string) (Nickname, error) {
	if !nicknamePattern.MatchString(raw) {
		return Nickname{}, ErrUserInvalidNickname
	}
	return Nickname{value: raw}, nil
}

// RestoreNickname trusts a stored value, so a rule tightened later never stops an old row from loading.
func RestoreNickname(value string) Nickname {
	return Nickname{value: value}
}

func (n Nickname) String() string {
	return n.value
}
```

## internal/modules/user/domain/user.go

```go
package domain

import "time"

type User struct {
	id          string
	nickname    Nickname
	status      UserStatus
	createdAt   time.Time
	withdrawnAt *time.Time
}

type UserSnapshot struct {
	ID          string
	Nickname    string
	Status      UserStatus
	CreatedAt   time.Time
	WithdrawnAt *time.Time
}

type RegisterUserInput struct {
	ID       string
	Nickname string
}

func RegisterUser(in RegisterUserInput, now time.Time) (*User, error) {
	nickname, err := NewNickname(in.Nickname)
	if err != nil {
		return nil, err
	}
	return &User{id: in.ID, nickname: nickname, status: UserStatusActive, createdAt: now}, nil
}

func RestoreUser(s UserSnapshot) *User {
	return &User{
		id:          s.ID,
		nickname:    RestoreNickname(s.Nickname),
		status:      s.Status,
		createdAt:   s.CreatedAt,
		withdrawnAt: s.WithdrawnAt,
	}
}

func (u *User) Snapshot() UserSnapshot {
	return UserSnapshot{
		ID:          u.id,
		Nickname:    u.nickname.String(),
		Status:      u.status,
		CreatedAt:   u.createdAt,
		WithdrawnAt: u.withdrawnAt,
	}
}

func (u *User) ID() string {
	return u.id
}

func (u *User) Withdraw(now time.Time) error {
	if u.status == UserStatusWithdrawn {
		return ErrUserAlreadyWithdrawn
	}
	u.status = UserStatusWithdrawn
	u.withdrawnAt = &now
	return nil
}
```

## internal/modules/user/application/withdraw_user.go

```go
package application

import (
	"context"
	"time"

	"example.com/server/internal/modules/user/domain"
)

type WithdrawUserCommand struct {
	UserID string
	Reason domain.WithdrawalReason
}

type WithdrawUser struct {
	users       domain.UserRepository
	revoker     AccessRevoker
	transaction TransactionRunner
	now         func() time.Time
}

func NewWithdrawUser(users domain.UserRepository, revoker AccessRevoker, transaction TransactionRunner, now func() time.Time) *WithdrawUser {
	return &WithdrawUser{users: users, revoker: revoker, transaction: transaction, now: now}
}

func (uc *WithdrawUser) Execute(ctx context.Context, cmd WithdrawUserCommand) error {
	return uc.transaction.Run(ctx, func(ctx context.Context) error {
		u, err := uc.users.FindByID(ctx, cmd.UserID)
		if err != nil {
			return err
		}
		if err := u.Withdraw(uc.now()); err != nil {
			return err
		}
		if err := uc.users.Save(ctx, u); err != nil {
			return err
		}
		return uc.revoker.RevokeAll(ctx, cmd.UserID)
	})
}
```

## internal/modules/user/adapters/out/persistence/mapper.go

```go
package persistence

import (
	"fmt"

	"github.com/google/uuid"

	"example.com/server/internal/modules/user/adapters/out/persistence/sqlcgen"
	"example.com/server/internal/modules/user/domain"
)

func toDomain(row sqlcgen.User) *domain.User {
	return domain.RestoreUser(domain.UserSnapshot{
		ID:          row.ID.String(),
		Nickname:    row.Nickname,
		Status:      domain.UserStatus(row.Status),
		CreatedAt:   row.CreatedAt,
		WithdrawnAt: row.WithdrawnAt,
	})
}

func toUpsertParams(u *domain.User) (sqlcgen.UpsertUserParams, error) {
	s := u.Snapshot()
	id, err := uuid.Parse(s.ID)
	if err != nil {
		return sqlcgen.UpsertUserParams{}, fmt.Errorf("parse user id %s: %w", s.ID, err)
	}
	return sqlcgen.UpsertUserParams{
		ID:          id,
		Nickname:    s.Nickname,
		Status:      sqlcgen.UserStatus(s.Status),
		CreatedAt:   s.CreatedAt,
		WithdrawnAt: s.WithdrawnAt,
	}, nil
}

func toNullStatus(status *domain.UserStatus) sqlcgen.NullUserStatus {
	if status == nil {
		return sqlcgen.NullUserStatus{}
	}
	return sqlcgen.NullUserStatus{UserStatus: sqlcgen.UserStatus(*status), Valid: true}
}
```

## internal/modules/user/adapters/out/persistence/user_repository.go

```go
package persistence

import (
	"context"
	"errors"
	"fmt"

	"github.com/google/uuid"
	"github.com/jackc/pgx/v5"

	"example.com/server/internal/infrastructure/database"
	"example.com/server/internal/modules/user/adapters/out/persistence/sqlcgen"
	"example.com/server/internal/modules/user/domain"
	"example.com/server/internal/shared/page"
)

type UserRepository struct {
	db *database.DB
}

var _ domain.UserRepository = (*UserRepository)(nil)

func NewUserRepository(db *database.DB) *UserRepository {
	return &UserRepository{db: db}
}

func (r *UserRepository) FindByID(ctx context.Context, id string) (*domain.User, error) {
	uid, err := uuid.Parse(id)
	if err != nil {
		return nil, domain.ErrUserNotFound // a malformed id names no stored user
	}
	row, err := sqlcgen.New(r.db.Conn(ctx)).GetUser(ctx, uid)
	if errors.Is(err, pgx.ErrNoRows) {
		return nil, domain.ErrUserNotFound
	}
	if err != nil {
		return nil, fmt.Errorf("get user %s: %w", id, err)
	}
	return toDomain(row), nil
}

func (r *UserRepository) FindPage(ctx context.Context, c domain.UserListCriteria) (page.Result[*domain.User], error) {
	q := sqlcgen.New(r.db.Conn(ctx))
	status := toNullStatus(c.Status)
	rows, err := q.ListUsers(ctx, sqlcgen.ListUsersParams{
		Status:     status,
		PageSize:   int64(c.PageSize),
		PageOffset: int64((c.Page - 1) * c.PageSize),
	})
	if err != nil {
		return page.Result[*domain.User]{}, fmt.Errorf("list users: %w", err)
	}
	total, err := q.CountUsers(ctx, status)
	if err != nil {
		return page.Result[*domain.User]{}, fmt.Errorf("count users: %w", err)
	}
	users := make([]*domain.User, 0, len(rows))
	for _, row := range rows {
		users = append(users, toDomain(row))
	}
	return page.Result[*domain.User]{Items: users, Total: int(total), Page: c.Page, PageSize: c.PageSize}, nil
}

func (r *UserRepository) Save(ctx context.Context, u *domain.User) error {
	params, err := toUpsertParams(u)
	if err != nil {
		return err
	}
	err = sqlcgen.New(r.db.Conn(ctx)).UpsertUser(ctx, params)
	if database.IsUniqueViolation(err, "users_nickname_key") {
		return domain.ErrUserNicknameTaken
	}
	if err != nil {
		return fmt.Errorf("save user %s: %w", u.ID(), err)
	}
	return nil
}
```

```go
	t.Run("turns a nickname another user has into ErrUserNicknameTaken", func(t *testing.T) {
```

```go
package httpapi

import (
	"context"
	"net/http"

	"github.com/danielgtaylor/huma/v2"

	"example.com/server/internal/common/apiresponse"
	"example.com/server/internal/modules/user/application"
	"example.com/server/internal/modules/user/domain"
)

type Handler struct {
	getUserList   *application.GetUserList
	getUserDetail *application.GetUserDetail
	withdrawUser  *application.WithdrawUser
}

func NewHandler(getUserList *application.GetUserList, getUserDetail *application.GetUserDetail, withdrawUser *application.WithdrawUser) *Handler {
	return &Handler{getUserList: getUserList, getUserDetail: getUserDetail, withdrawUser: withdrawUser}
}

func (h *Handler) Register(api huma.API) {
	huma.Register(api, huma.Operation{
		OperationID: "get-user-list", Method: http.MethodGet, Path: "/api/users",
		Summary: "List users", Tags: []string{"users"},
	}, h.getList)
	huma.Register(api, huma.Operation{
		OperationID: "get-user-detail", Method: http.MethodGet, Path: "/api/users/{id}",
		Summary: "Get a user", Tags: []string{"users"},
	}, h.getDetail)
	huma.Register(api, huma.Operation{
		OperationID: "withdraw-user", Method: http.MethodPost, Path: "/api/users/{id}/withdraw",
		Summary: "Withdraw a user", Tags: []string{"users"}, DefaultStatus: http.StatusOK,
	}, h.withdraw)
}

func (h *Handler) getList(ctx context.Context, in *UserListParams) (*apiresponse.Output[apiresponse.Page[UserSummaryResponse]], error) {
	var status *domain.UserStatus
	if in.Status != "" {
		s := domain.UserStatus(in.Status)
		status = &s
	}
	result, err := h.getUserList.Execute(ctx, application.GetUserListQuery{Status: status, Page: in.Page, PageSize: in.PageSize})
	if err != nil {
		return nil, err
	}
	return apiresponse.OK(ctx, apiresponse.FromPage(result, NewUserSummaryResponse)), nil
}

func (h *Handler) getDetail(ctx context.Context, in *GetUserDetailRequest) (*apiresponse.Output[UserResponse], error) {
	result, err := h.getUserDetail.Execute(ctx, application.GetUserDetailQuery{UserID: in.ID})
	if err != nil {
		return nil, err
	}
	return apiresponse.OK(ctx, NewUserResponse(result)), nil
}

func (h *Handler) withdraw(ctx context.Context, in *WithdrawUserRequest) (*apiresponse.Output[*struct{}], error) {
	err := h.withdrawUser.Execute(ctx, application.WithdrawUserCommand{
		UserID: in.ID,
		Reason: domain.WithdrawalReason(in.Body.Reason),
	})
	if err != nil {
		return nil, err
	}
	return apiresponse.Empty(ctx), nil
}
```
